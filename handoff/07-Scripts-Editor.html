1# mora_diaria.py — recalcula días en mora, mora S/ y saldo restante
2# Corre cada noche y al registrar pagos. NO editar a mano.
3
4from opsgrid import ds, today, days_between
5
6def compute(row):
7 # 1. Días entre hoy y la fecha de vencimiento
8 dias = days_between(today(), row.vence)
9 dias_mora = max(0, dias)
10
11 # 2. Tasa de mora desde la tabla relacionada Clientes
12 tasa = row.cliente.tasa_mora # lookup automático
13 mora_pen = row.cuota * tasa * (dias_mora / 30)
14
15 # 3. Saldo restante = monto - pagos cargados a este préstamo
16 pagado = ds.sum("Pagos", where=lambda p: p.prestamo == row.id, col="monto")
17 saldo = row.monto - pagado
18
19 return {
20 "dias_mora": dias_mora,
21 "mora_pen": round(mora_pen, 2),
22 "saldo": round(saldo, 2),
23 }
